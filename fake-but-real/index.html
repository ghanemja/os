        Privacy-safe by design
        
Nobody gets a call. Nobody gets an email.

        
Demo data leaks: into screenshots, staging databases, test emails that actually send. fake-but-real only uses phone numbers and domains that are reserved for fiction and documentation, so a stray "Send" goes nowhere.

        Get started
        
        One import. No dependencies.
        
        
        Works in Node 18+ and every modern browser. TypeScript types are included.
        
        
        
          
        
            1 · Install
            
        From GitHub
        
            
        npm i github:ghanemja/fake-but-real
        
            
        Or copy index.js into your project. It is one file.
        
          
        
          
        
            2 · Generate
            
        Seed it, call it
        
            
        import { createFaker } from 'fake-but-real';

const fake = createFaker('team-page');
fake.person();   // { fullName, email, phone, ... }
fake.product();  // { name, price, currency }
fake.review();   // { rating, title, text, author }
        
          
        
          
        
            3 · Fill a mockup
            
        A whole team page
        
            
        const team = Array.from({ length: 12 }, () => {
  const p = fake.person();
  return { ...p, title: fake.jobTitle(),
    bio: fake.bio({ firstName: p.firstName }) };
});
        
          
        
          
        
            4 · Stable tests
            
        Snapshot-safe fixtures
        
            
        import { test } from 'node:test';
import assert from 'node:assert/strict';
import { createFaker } from 'fake-but-real';

test('renders a user card', () => {
  const user = createFaker(42).person();
  // same user on every run and every machine
  assert.match(user.email, /@example\.(com|org|net)$/);
});
